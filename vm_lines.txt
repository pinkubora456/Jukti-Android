   119	    private fun syncTrustedTime() {
   120	        viewModelScope.launch(kotlinx.coroutines.Dispatchers.IO) {
   121	            try {
   122	                // Fetch header from a reliable Google endpoint
   123	                val url = java.net.URL("https://us-central1-jukti-examprep.cloudfunctions.net")
   124	                val conn = url.openConnection() as java.net.HttpURLConnection
   125	                conn.requestMethod = "HEAD"
   126	                conn.connectTimeout = 3000
   127	                conn.readTimeout = 3000
   128	                val dateStr = conn.getHeaderField("Date")
   129	                if (dateStr != null) {
   130	                    val format = java.text.SimpleDateFormat("EEE, dd MMM yyyy HH:mm:ss zzz", java.util.Locale.US)
   131	                    val serverTime = format.parse(dateStr)?.time ?: return@launch
   132	                    
   133	                    sessionTrustedServerTime = serverTime
   134	                    sessionTrustedRealtime = android.os.SystemClock.elapsedRealtime()
   135	                    
   136	                    timePrefs.edit()
   137	                        .putLong("last_server_time", serverTime)
   138	                        .putLong("last_device_realtime", sessionTrustedRealtime)
   139	                        .putLong("max_seen_time", serverTime)
   140	                        .apply()
   141	                }
   142	            } catch (e: Exception) {
   143	                // Ignore, will fallback to cached or monotonic
   144	            }
   145	        }
   146	    }
   147	
   148	    fun getTrustedTime(): Long {
   149	        val currentRealtime = android.os.SystemClock.elapsedRealtime()
   150	        
   151	        // 1. If we have a session-synced server time, use it + monotonic elapsed time
   152	        if (sessionTrustedServerTime > 0L) {
   153	            val elapsed = currentRealtime - sessionTrustedRealtime
   154	            if (elapsed >= 0) {
   155	                val calculated = sessionTrustedServerTime + elapsed
   156	                updateMaxSeenTime(calculated)
   157	                return calculated
   158	            }
   159	        }
   160	        
   161	        // 2. Fallback to persisted synced time if device hasn't rebooted since last sync
   162	        val lastRealtime = timePrefs.getLong("last_device_realtime", 0L)
   163	        val lastServerTime = timePrefs.getLong("last_server_time", 0L)
   164	        if (lastServerTime > 0L && lastRealtime > 0L && currentRealtime >= lastRealtime) {
   165	            val elapsed = currentRealtime - lastRealtime
   166	            // basic sanity check: if elapsed is > 30 days without reboot, maybe suspicious, but let's trust it
   167	            if (elapsed < 30L * 24 * 60 * 60 * 1000) {
   168	                val calculated = lastServerTime + elapsed
   169	                updateMaxSeenTime(calculated)
   170	                return calculated
   171	            }
   172	        }
   173	        
   174	        // 3. Last resort fallback: System time, but protected against backwards tampering
   175	        val currentSystemTime = System.currentTimeMillis()
   176	        return updateMaxSeenTime(currentSystemTime)
   177	    }
   178	    
   179	    private fun updateMaxSeenTime(time: Long): Long {
   180	        val maxSeen = timePrefs.getLong("max_seen_time", 0L)
   181	        if (time > maxSeen) {
   182	            timePrefs.edit().putLong("max_seen_time", time).apply()
   183	            return time
   184	        }
   185	        return maxSeen
   186	    }

   637	    suspend fun validatePurchaseEligibility(plan: com.example.data.local.PlanEntity): Pair<Boolean, String> {
   638	        val prof = userProfile.value
   639	        if (prof != null && prof.email.isNotBlank()) {
   640	            val docId = com.example.data.repository.FirebaseRepository().getSanitizedUserDocId(prof.email)
   641	            try {
   642	                val db = com.example.JuktiApplication.getFirestore(getApplication())
   643	                    ?: return Pair(false, "Firebase unavailable")
   644	                val doc = db.collection("users").document(docId)
   645	                    .collection("entitlements").document("current").get().await()
   646	                if (doc.exists()) {
   647	                    val status = doc.getString("status") ?: ""
   648	                    val planId = doc.getString("planId") ?: ""
   649	                    val planName = doc.getString("planName") ?: ""
   650	                    val validUntil = doc.getLong("validUntil") ?: 0L
   651	
   652	                    val isMatch = (planId == plan.id.toString()) ||
   653	                                  (planId.equals(plan.planName, ignoreCase = true)) ||
   654	                                  (planName.equals(plan.planName, ignoreCase = true))
   655	                    val isValid = validUntil <= 0L || validUntil > getTrustedTime()
   656	
   657	                    if (status == "ACTIVE" && isMatch && isValid) {
   658	                        return Pair(false, "You already have an active subscription for '${plan.planName}'. Duplicate purchases of the same plan are not allowed.")
   659	                    }
   660	                }
   661	            } catch (e: Exception) {
   662	                // Fallback to local check if offline
   663	            }
   664	        }
   665	
   666	        if (isSpecificPlanActive(plan)) {
   667	            return Pair(false, "You already have an active subscription for '${plan.planName}'.")
   668	        }
   669	
   670	        return Pair(true, "")
   671	    }

  1238	    private fun serializeMockAttemptAnswers(
  1239	        answers: Map<Int, Int>,
  1240	        questionIds: List<Long>,
  1241	        timeSpentSeconds: Int,
  1242	        rawScore: Float
  1243	    ): String {
  1244	        val root = org.json.JSONObject()
  1245	        val answersObj = org.json.JSONObject()
  1246	        answers.forEach { (idx, opt) ->
  1247	            val qId = questionIds.getOrNull(idx)
  1248	            answersObj.put(idx.toString(), opt)
  1249	            if (qId != null) {

  1251	            }

  1253	        root.put("answers", answersObj)
  1254	        root.put("timeSpentSeconds", timeSpentSeconds)

  1259	    private fun deserializeMockAttemptAnswers(
  1260	        jsonStr: String,
  1261	        questionIds: List<Long>
  1262	    ): Pair<Map<Int, Int>, Int> {
  1263	        if (jsonStr.isBlank()) return Pair(emptyMap(), 0)
  1264	        val map = mutableMapOf<Int, Int>()
  1265	        var timeSpent = 0
  1266	        try {
  1267	            val root = org.json.JSONObject(jsonStr)
  1268	            if (root.has("answers")) {
  1269	                val answersObj = root.getJSONObject("answers")
  1270	                questionIds.forEachIndexed { index, qId ->
  1271	                    if (answersObj.has(index.toString())) {
  1272	                        map[index] = answersObj.getInt(index.toString())
  1273	                    } else if (answersObj.has("qid_$qId")) {
  1274	                        map[index] = answersObj.getInt("qid_$qId")
  1275	                    }
  1276	                }
  1277	                timeSpent = root.optInt("timeSpentSeconds", 0)
  1278	            } else {
  1279	                val keys = root.keys()
  1280	                while (keys.hasNext()) {
  1281	                    val k = keys.next()
  1282	                    val idx = k.toIntOrNull()
  1283	                    if (idx != null) {
  1284	                        map[idx] = root.getInt(k)
  1285	                    }
  1286	                }
  1287	            }
  1288	        } catch (e: Exception) {
  1289	            // Ignore format errors
  1290	        }
  1291	        return Pair(map, timeSpent)
  1292	    }

  1535	    fun submitCurrentMockTest() {
  1536	        if (_isSubmittingMock.value) return
  1537	        val test = _selectedMockTest.value ?: return
  1538	        val currentQuestionsList = _activeMockQuestions.value
  1539	        if (currentQuestionsList.isEmpty()) return
  1540

  1547	        var qMarksMap: org.json.JSONObject? = null

  1550	                qMarksMap = org.json.JSONObject(test.questionMarksJson)
  1551	            }
  1552	        } catch (e: Exception) {}
  1553	        
  1554	        val markPerQuestion = if (test.markPerQuestion > 0f) test.markPerQuestion else (test.totalMarks / totalQ.coerceAtLeast(1).toFloat())
  1555	        val negPerQuestion = parseNegativeMarkingRate(test.negativeMarking)
  1556	
  1557	        var correctCount = 0
  1558	        var incorrectCount = 0
  1559	        var skippedCount = 0
  1560	        var earnedMarks = 0f
  1561	        var deductedMarks = 0f
  1562	
  1563	        val userId = userProfile.value?.uid?.ifBlank { "local_user" } ?: "local_user"
  1564	
  1565	        currentQuestionsList.forEachIndexed { index, q ->
  1566	            val userChoice = answers[index]
  1567	            var actualMarkForQ = markPerQuestion
  1568	            try {
  1569	                if (qMarksMap != null && qMarksMap.has(q.id.toString())) {
  1570	                    actualMarkForQ = qMarksMap.getDouble(q.id.toString()).toFloat()
  1571	                }
  1572	            } catch (e: Exception) {}
  1573	            
  1574	            when {
  1575	                userChoice == q.correctOptionIndex -> {
  1576	                    correctCount++
  1577	                    earnedMarks += actualMarkForQ
  1578	                    viewModelScope.launch {
  1579	                        try {
  1580	                            repository.recordQuestionAnswer(userId = userId, questionId = q.id.toString(), isCorrect = true)
  1581	                        } catch (e: Exception) {}
  1582	                    }
  1583	                }
  1584	                userChoice != null -> {
  1585	                    incorrectCount++
  1586	                    deductedMarks += negPerQuestion
  1587	                    viewModelScope.launch {
  1588	                        try {
  1589	                            repository.recordQuestionAnswer(userId = userId, questionId = q.id.toString(), isCorrect = false)

  1592	                }
  1593	                else -> {
  1594	                    skippedCount++
  1595	                }
  1596	            }
  1597	        }
  1598	
  1599	        val totalAttempted = correctCount + incorrectCount
  1600	        val rawScore = earnedMarks - deductedMarks
  1601	        val finalScore = rawScore.coerceAtLeast(0f)
  1602	        val accuracy = if (totalAttempted > 0) ((correctCount.toFloat() / totalAttempted.toFloat()) * 100f).coerceIn(0f, 100f) else 0f
  1603	        val timeSpentSec = ((test.durationMinutes * 60) - _mockTimeRemainingSeconds.value).coerceAtLeast(1)
  1604	
  1605	        val questionIdsList = currentQuestionsList.map { it.id }
  1606	        val answersJson = serializeMockAttemptAnswers(answers, questionIdsList, timeSpentSec, rawScore)

  1611	            userId = userId,
  1612	            timestamp = System.currentTimeMillis(),

  1614	            userAnswersJson = answersJson,
  1615	            score = finalScore,
  1616	            totalMarks = test.totalMarks,

  1618	            correctCount = correctCount,
  1619	            totalAttempted = totalAttempted,
  1620	            questionMarksJson = test.questionMarksJson
  1621	        )

  1628	                _currentMockAttempt.value = savedAttempt
  1629	
  1630	                // 2. Compute dynamic rank & percentile against all completed attempts
  1631	                val attemptsForMock = repository.getAllAttemptsForMock(test.id).firstOrNull() ?: emptyList()
  1632	                val scorePercentage = if (test.totalMarks > 0f) ((finalScore.toFloat() / test.totalMarks.toFloat()) * 100f) else 0f
  1633	                val dynamicPercentile = if (attemptsForMock.size > 1) {
  1634	                    val lowerScoreCount = attemptsForMock.count { it.score < finalScore }
  1635	                    ((lowerScoreCount.toFloat() / (attemptsForMock.size - 1).toFloat()) * 100f).coerceIn(5.0f, 99.9f)
  1636	                } else {
  1637	                    (scorePercentage * 0.7f + accuracy * 0.3f).coerceIn(5.0f, 99.9f)
  1638	                }
  1639	
  1640	                val rank = if (attemptsForMock.isNotEmpty()) {
  1641	                    val higherCount = attemptsForMock.count { it.score > finalScore }
  1642	                    higherCount + 1
  1643	                } else 1
  1644	
  1645	                // 3. Update mock test in database with submission results
  1646	                repository.submitMockResult(
  1647	                    mockId = test.id,
  1648	                    score = finalScore,
  1649	                    accuracy = accuracy,
  1650	                    timeSpentMins = (timeSpentSec / 60).coerceAtLeast(1),
  1651	                    totalAttempted = totalAttempted,
  1652	
  1653	                    correctCount = correctCount
  1654	                )
  1655	
  1656	                _selectedMockTest.value = test.copy(
  1657	                    isCompleted = true,
  1658	                    userScore = finalScore,
  1659	                    userAccuracy = accuracy,
  1660	                    userRank = rank,

  2501	    private fun parseCsvLine(line: String): List<String> {
  2502	        val result = mutableListOf<String>()
  2503	        var current = StringBuilder()
  2504	        var inQuotes = false
  2505	        for (i in line.indices) {
  2506	            val char = line[i]
  2507	            if (char == '\"') {
  2508	                inQuotes = !inQuotes
  2509	            } else if (char == ',' && !inQuotes) {
  2510	                result.add(current.toString().trim())
  2511	                current.clear()
  2512	            } else {
  2513	                current.append(char)
  2514	            }
  2515	        }
  2516	        result.add(current.toString().trim())
  2517	        return result
  2518	    }

  3283	    private fun saveCsvToFile(context: android.content.Context, filename: String, content: String) {
  3284	        try {
  3285	            var savedUri: android.net.Uri? = null
  3286	            if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.Q) {
  3287	                val resolver = context.contentResolver
  3288	                val contentValues = android.content.ContentValues().apply {
  3289	                    put(android.provider.MediaStore.MediaColumns.DISPLAY_NAME, filename)
  3290	                    put(android.provider.MediaStore.MediaColumns.MIME_TYPE, "text/csv")
  3291	                    put(android.provider.MediaStore.MediaColumns.RELATIVE_PATH, android.os.Environment.DIRECTORY_DOWNLOADS)
  3292	                }
  3293	                savedUri = resolver.insert(android.provider.MediaStore.Downloads.EXTERNAL_CONTENT_URI, contentValues)
  3294	                if (savedUri != null) {
  3295	                    resolver.openOutputStream(savedUri)?.use { it.write(content.toByteArray()) }
  3296	                }
  3297	            } else {
  3298	                val dir = android.os.Environment.getExternalStoragePublicDirectory(android.os.Environment.DIRECTORY_DOWNLOADS)
  3299	                if (!dir.exists()) dir.mkdirs()
  3300	                val file = java.io.File(dir, filename)
  3301	                java.io.FileOutputStream(file).use { it.write(content.toByteArray()) }
  3302	                savedUri = android.net.Uri.fromFile(file)
  3303	            }
  3304	
  3305	            launchOnMain {
  3306	                android.widget.Toast.makeText(
  3307	                    context,
  3308	                    "Saved $filename to Downloads folder!",
  3309	                    android.widget.Toast.LENGTH_LONG
  3310	                ).show()
  3311	
  3312	                if (savedUri != null) {
  3313	                    try {
  3314	                        val shareIntent = android.content.Intent(android.content.Intent.ACTION_SEND).apply {
  3315	                            type = "text/csv"
  3316	                            putExtra(android.content.Intent.EXTRA_STREAM, savedUri)
  3317	                            addFlags(android.content.Intent.FLAG_GRANT_READ_URI_PERMISSION)
  3318	                        }
  3319	                        val chooser = android.content.Intent.createChooser(shareIntent, "Open or Share $filename")
  3320	                        chooser.addFlags(android.content.Intent.FLAG_ACTIVITY_NEW_TASK)
  3321	                        context.startActivity(chooser)
  3322	                    } catch (e: Exception) {
  3323	                        android.util.Log.e("JuktiViewModel", "Share intent failed", e)
  3324	                    }
  3325	                }
  3326	            }
  3327	        } catch (e: Exception) {
  3328	            android.util.Log.e("JuktiViewModel", "Error saving CSV", e)
  3329	            launchOnMain {
  3330	                android.widget.Toast.makeText(context, "Failed to save report: ${e.localizedMessage}", android.widget.Toast.LENGTH_SHORT).show()
  3331	            }
  3332	        }
  3333	    }
